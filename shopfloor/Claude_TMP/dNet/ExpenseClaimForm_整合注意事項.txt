```vb.net
Private Sub SaveToJtdb_jMGUIAP(jID As Integer, details As DataTable)
    Dim docTotal As Decimal = CalculateDocTotal(details)
    Dim vatSum As Decimal = CalculateVatSum(details)

    Dim sql As String = "
        INSERT INTO jMGUIAP (
            jID, DocTotal, VatSum, U_OBJTYPE,
            MDRPostStatus, CreateDate, CreateBy
        ) VALUES (
            @jID, @DocTotal, @VatSum, '18',
            'N', GETDATE(), @CreateBy
        )
    "

    Dim cmd As New SqlCommand()
    cmd.Parameters.AddWithValue("@jID", jID)
    cmd.Parameters.AddWithValue("@DocTotal", docTotal)
    cmd.Parameters.AddWithValue("@VatSum", vatSum)
    cmd.Parameters.AddWithValue("@CreateBy", CommUtil.GetCurrentUserId())

    ' 使用稽核欄位方法（已整合）

End Sub
```

#### UpdateJtdb_DocEntry()
```vb.net
''' <summary>
''' 更新 jMGUIAP 的 DocEntry（AP Invoice 建立成功後）
''' </summary>
Private Sub UpdateJtdb_DocEntry(jID As Integer, docEntry As Integer)
    Dim sql As String = "
        UPDATE jMGUIAP
        SET DocEntry = @DocEntry,
            DocNum = @DocNum,
            UpdateDate = GETDATE(),
            UpdateBy = @UpdateBy
        WHERE jID = @jID
    "

    Dim cmd As New SqlCommand()
    cmd.Parameters.AddWithValue("@DocEntry", docEntry)
    cmd.Parameters.AddWithValue("@DocNum", docEntry)  ' 假設 DocNum = DocEntry
    cmd.Parameters.AddWithValue("@jID", jID)
    cmd.Parameters.AddWithValue("@UpdateBy", CommUtil.GetCurrentUserId())

    CommUtil.ExecuteUpdateWithAudit(sql, cmd, LocalConn)
End Sub
```
