一、用途說明
---------------------------------------------------------------------
本檔案包含費用申請單的 MDR（營業稅發票）資料同步功能程式碼。
當費用申請單審核通過並成功產生 SAP B1 AP Invoice 後，需將營業稅
發票資料同步到 MDR 資料庫，供 MDRImport.exe 程式處理。

二、目標位置
---------------------------------------------------------------------
應加入到：MgmSP/ExpenseClaimForm.aspx.vb（或相關的 AP Invoice 表單）
位置：Code-Behind 檔案中，建議放在儲存成功後的流程

三、功能說明
---------------------------------------------------------------------
1. WriteMDRData()
   - 從 jtdb 讀取 jMGUIAP 和 jMGUIAPDetail 資料
   - 寫入 MDR.MGUIAP_Import（表頭）
   - 寫入 MDR.MGUIAPDetail_Import（明細）
   - 更新 jMGUIAP.MDRPostStatus 狀態
   - 回傳 MDR.MGUIAP_Import.ID

2. CallMDRProgram()
   - 呼叫 MDRImport.exe 執行營業稅過帳
   - 傳遞 jID 參數
   - （本版本為保留功能，已註解）

四、使用方式
---------------------------------------------------------------------
在 AP Invoice 建立成功後呼叫：

' 儲存成功且產生 AP Invoice
If docEntry > 0 Then
    Try
        ' 同步到 MDR 資料庫
        Dim mdrHeaderID As Integer = WriteMDRData(jID)

        If mdrHeaderID > 0 Then
            ' MDR 資料寫入成功
            ' [保留] 呼叫 MDRImport.exe 程式
            ' CallMDRProgram(jID)

            Response.Write("<script>alert('儲存成功！AP Invoice DocEntry: " & docEntry & "');</script>")
        Else
            Response.Write("<script>alert('儲存成功，但 MDR 同步失敗，請檢查日誌');</script>")
        End If
    Catch ex As Exception
        ' MDR 同步失敗不影響主流程
        Response.Write("<script>alert('儲存成功，但 MDR 同步發生錯誤：" & ex.Message & "');</script>")
    End Try
End If

#Region "MDR 資料同步（營業稅發票）"

''' <summary>
''' 將營業稅發票資料同步到 MDR 資料庫
''' </summary>
''' <param name="jID">費用申請單 jID</param>
''' <returns>MDR.MGUIAP_Import.ID，失敗回傳 0</returns>
''' <remarks>
''' 此方法會從 jtdb 讀取 jMGUIAP 和 jMGUIAPDetail 資料，
''' 寫入 MDR 資料庫的 MGUIAP_Import 和 MGUIAPDetail_Import 表，
''' 供後續 MDRImport.exe 程式處理營業稅過帳。
''' </remarks>
Private Function WriteMDRData(jID As Integer) As Integer
    Dim mdrHeaderID As Integer = 0
    Dim jtdbConn As SqlConnection = Nothing
    Dim mdrConn As SqlConnection = Nothing
    Dim trans As SqlTransaction = Nothing

    Try
        ' ===== 第一步：從 jtdb 讀取資料 =====
        Dim headerData As DataRow = Nothing
        Dim detailData As DataTable = Nothing

        jtdbConn = New SqlConnection(ConfigurationManager.ConnectionStrings("jtdbConnectionString").ConnectionString)
        jtdbConn.Open()

        ' 讀取表頭資料
        Dim sqlHeader As String = "
            SELECT
                h.jID,
                h.DocEntry,
                h.DocNum,
                d.U_LIFNR,
                d.U_STCEG,
                d.U_XBLNR,
                d.U_BLDAT,
                d.U_VATDATE,
                d.U_ZFORM_CODE,
                d.U_TAX_TYPE,
                d.U_CUS_TYPE,
                d.U_AM_TYPE,
                d.U_BUKRS,
                d.U_MWSKZ,
                d.U_VATCODE,
                SUM(d.U_HWBAS) AS U_HWBAS,
                SUM(d.U_HWSTE) AS U_HWSTE,
                SUM(d.U_HWBAS + d.U_HWSTE) AS TotalAmount
            FROM jMGUIAP h
            INNER JOIN jMGUIAPDetail d ON h.jID = d.jID
            WHERE h.jID = @jID
            GROUP BY
                h.jID, h.DocEntry, h.DocNum,
                d.U_LIFNR, d.U_STCEG, d.U_XBLNR,
                d.U_BLDAT, d.U_VATDATE, d.U_ZFORM_CODE,
                d.U_TAX_TYPE, d.U_CUS_TYPE, d.U_AM_TYPE,
                d.U_BUKRS, d.U_MWSKZ, d.U_VATCODE
        "

        Dim cmdHeader As New SqlCommand(sqlHeader, jtdbConn)
        cmdHeader.Parameters.AddWithValue("@jID", jID)
        Dim adapterHeader As New SqlDataAdapter(cmdHeader)
        Dim dtHeader As New DataTable()
        adapterHeader.Fill(dtHeader)

        If dtHeader.Rows.Count = 0 Then
            Throw New Exception("找不到 jID=" & jID & " 的發票表頭資料")
        End If
        headerData = dtHeader.Rows(0)

        ' 讀取明細資料
        Dim sqlDetail As String = "
            SELECT
                jID,
                LineNum,
                DocEntry,
                U_LIFNR,
                U_STCEG,
                U_XBLNR,
                U_ZFORM_CODE,
                U_BLDAT,
                U_VATDATE,
                U_HWBAS,
                U_HWSTE,
                U_TAX_TYPE,
                U_CUS_TYPE,
                U_AM_TYPE,
                U_VATCODE,
                U_BUKRS,
                U_MWSKZ,
                U_BELNR,
                U_FA_DESC,
                U_FA_QTY,
                U_FA_USE,
                U_GatherMark,
                U_ConsolidQty
            FROM jMGUIAPDetail
            WHERE jID = @jID
            ORDER BY LineNum
        "

        Dim cmdDetail As New SqlCommand(sqlDetail, jtdbConn)
        cmdDetail.Parameters.AddWithValue("@jID", jID)
        Dim adapterDetail As New SqlDataAdapter(cmdDetail)
        detailData = New DataTable()
        adapterDetail.Fill(detailData)

        If detailData.Rows.Count = 0 Then
            Throw New Exception("找不到 jID=" & jID & " 的發票明細資料")
        End If

        ' ===== 第二步：寫入 MDR 資料庫 =====
        mdrConn = New SqlConnection(ConfigurationManager.ConnectionStrings("MDRConnectionString").ConnectionString)
        mdrConn.Open()
        trans = mdrConn.BeginTransaction()

        ' 寫入表頭
        Dim sqlInsertHeader As String = "
            INSERT INTO MGUIAP_Import (
                jID, DocEntry, DocNum,
                U_LIFNR, U_STCEG,
                U_XBLNR, U_BLDAT, U_VATDATE,
                U_HWBAS, U_HWSTE, TotalAmount,
                U_ZFORM_CODE, U_TAX_TYPE, U_CUS_TYPE, U_AM_TYPE,
                U_BUKRS, U_MWSKZ, U_VATCODE,
                PostStatus, CreateDate, CreateBy
            ) VALUES (
                @jID, @DocEntry, @DocNum,
                @U_LIFNR, @U_STCEG,
                @U_XBLNR, @U_BLDAT, @U_VATDATE,
                @U_HWBAS, @U_HWSTE, @TotalAmount,
                @U_ZFORM_CODE, @U_TAX_TYPE, @U_CUS_TYPE, @U_AM_TYPE,
                @U_BUKRS, @U_MWSKZ, @U_VATCODE,
                'N', GETDATE(), @CreateBy
            );
            SELECT SCOPE_IDENTITY();
        "

        Dim cmdInsertHeader As New SqlCommand(sqlInsertHeader, mdrConn, trans)
        cmdInsertHeader.Parameters.AddWithValue("@jID", jID)
        cmdInsertHeader.Parameters.AddWithValue("@DocEntry", If(IsDBNull(headerData("DocEntry")), DBNull.Value, headerData("DocEntry")))
        cmdInsertHeader.Parameters.AddWithValue("@DocNum", If(IsDBNull(headerData("DocNum")), DBNull.Value, headerData("DocNum")))
        cmdInsertHeader.Parameters.AddWithValue("@U_LIFNR", If(IsDBNull(headerData("U_LIFNR")), DBNull.Value, headerData("U_LIFNR")))
        cmdInsertHeader.Parameters.AddWithValue("@U_STCEG", If(IsDBNull(headerData("U_STCEG")), DBNull.Value, headerData("U_STCEG")))
        cmdInsertHeader.Parameters.AddWithValue("@U_XBLNR", If(IsDBNull(headerData("U_XBLNR")), DBNull.Value, headerData("U_XBLNR")))
        cmdInsertHeader.Parameters.AddWithValue("@U_BLDAT", If(IsDBNull(headerData("U_BLDAT")), DBNull.Value, headerData("U_BLDAT")))
        cmdInsertHeader.Parameters.AddWithValue("@U_VATDATE", If(IsDBNull(headerData("U_VATDATE")), DBNull.Value, headerData("U_VATDATE")))
        cmdInsertHeader.Parameters.AddWithValue("@U_HWBAS", headerData("U_HWBAS"))
        cmdInsertHeader.Parameters.AddWithValue("@U_HWSTE", headerData("U_HWSTE"))
        cmdInsertHeader.Parameters.AddWithValue("@TotalAmount", headerData("TotalAmount"))
        cmdInsertHeader.Parameters.AddWithValue("@U_ZFORM_CODE", If(IsDBNull(headerData("U_ZFORM_CODE")), DBNull.Value, headerData("U_ZFORM_CODE")))
        cmdInsertHeader.Parameters.AddWithValue("@U_TAX_TYPE", If(IsDBNull(headerData("U_TAX_TYPE")), DBNull.Value, headerData("U_TAX_TYPE")))
        cmdInsertHeader.Parameters.AddWithValue("@U_CUS_TYPE", If(IsDBNull(headerData("U_CUS_TYPE")), DBNull.Value, headerData("U_CUS_TYPE")))
        cmdInsertHeader.Parameters.AddWithValue("@U_AM_TYPE", If(IsDBNull(headerData("U_AM_TYPE")), DBNull.Value, headerData("U_AM_TYPE")))
        cmdInsertHeader.Parameters.AddWithValue("@U_BUKRS", If(IsDBNull(headerData("U_BUKRS")), DBNull.Value, headerData("U_BUKRS")))
        cmdInsertHeader.Parameters.AddWithValue("@U_MWSKZ", If(IsDBNull(headerData("U_MWSKZ")), DBNull.Value, headerData("U_MWSKZ")))
        cmdInsertHeader.Parameters.AddWithValue("@U_VATCODE", If(IsDBNull(headerData("U_VATCODE")), DBNull.Value, headerData("U_VATCODE")))
        cmdInsertHeader.Parameters.AddWithValue("@CreateBy", CommUtil.GetCurrentUserId())

        ' 執行並取得 IDENTITY
        mdrHeaderID = Convert.ToInt32(cmdInsertHeader.ExecuteScalar())

        ' 寫入明細
        For Each detailRow As DataRow In detailData.Rows
            Dim sqlInsertDetail As String = "
                INSERT INTO MGUIAPDetail_Import (
                    HeaderID, LineNum, jID, DocEntry,
                    U_LIFNR, U_STCEG, U_XBLNR, U_ZFORM_CODE,
                    U_BLDAT, U_VATDATE,
                    U_HWBAS, U_HWSTE,
                    U_TAX_TYPE, U_CUS_TYPE, U_AM_TYPE, U_VATCODE,
                    U_BUKRS, U_MWSKZ, U_BELNR,
                    U_FA_DESC, U_FA_QTY, U_FA_USE,
                    U_GatherMark, U_ConsolidQty,
                    CreateDate, CreateBy
                ) VALUES (
                    @HeaderID, @LineNum, @jID, @DocEntry,
                    @U_LIFNR, @U_STCEG, @U_XBLNR, @U_ZFORM_CODE,
                    @U_BLDAT, @U_VATDATE,
                    @U_HWBAS, @U_HWSTE,
                    @U_TAX_TYPE, @U_CUS_TYPE, @U_AM_TYPE, @U_VATCODE,
                    @U_BUKRS, @U_MWSKZ, @U_BELNR,
                    @U_FA_DESC, @U_FA_QTY, @U_FA_USE,
                    @U_GatherMark, @U_ConsolidQty,
                    GETDATE(), @CreateBy
                )
            "

            Dim cmdInsertDetail As New SqlCommand(sqlInsertDetail, mdrConn, trans)
            cmdInsertDetail.Parameters.AddWithValue("@HeaderID", mdrHeaderID)
            cmdInsertDetail.Parameters.AddWithValue("@LineNum", detailRow("LineNum"))
            cmdInsertDetail.Parameters.AddWithValue("@jID", detailRow("jID"))
            cmdInsertDetail.Parameters.AddWithValue("@DocEntry", If(IsDBNull(detailRow("DocEntry")), DBNull.Value, detailRow("DocEntry")))
            cmdInsertDetail.Parameters.AddWithValue("@U_LIFNR", If(IsDBNull(detailRow("U_LIFNR")), DBNull.Value, detailRow("U_LIFNR")))
            cmdInsertDetail.Parameters.AddWithValue("@U_STCEG", If(IsDBNull(detailRow("U_STCEG")), DBNull.Value, detailRow("U_STCEG")))
            cmdInsertDetail.Parameters.AddWithValue("@U_XBLNR", If(IsDBNull(detailRow("U_XBLNR")), DBNull.Value, detailRow("U_XBLNR")))
            cmdInsertDetail.Parameters.AddWithValue("@U_ZFORM_CODE", If(IsDBNull(detailRow("U_ZFORM_CODE")), DBNull.Value, detailRow("U_ZFORM_CODE")))
            cmdInsertDetail.Parameters.AddWithValue("@U_BLDAT", If(IsDBNull(detailRow("U_BLDAT")), DBNull.Value, detailRow("U_BLDAT")))
            cmdInsertDetail.Parameters.AddWithValue("@U_VATDATE", If(IsDBNull(detailRow("U_VATDATE")), DBNull.Value, detailRow("U_VATDATE")))
            cmdInsertDetail.Parameters.AddWithValue("@U_HWBAS", detailRow("U_HWBAS"))
            cmdInsertDetail.Parameters.AddWithValue("@U_HWSTE", detailRow("U_HWSTE"))
            cmdInsertDetail.Parameters.AddWithValue("@U_TAX_TYPE", If(IsDBNull(detailRow("U_TAX_TYPE")), DBNull.Value, detailRow("U_TAX_TYPE")))
            cmdInsertDetail.Parameters.AddWithValue("@U_CUS_TYPE", If(IsDBNull(detailRow("U_CUS_TYPE")), DBNull.Value, detailRow("U_CUS_TYPE")))
            cmdInsertDetail.Parameters.AddWithValue("@U_AM_TYPE", If(IsDBNull(detailRow("U_AM_TYPE")), DBNull.Value, detailRow("U_AM_TYPE")))
            cmdInsertDetail.Parameters.AddWithValue("@U_VATCODE", If(IsDBNull(detailRow("U_VATCODE")), DBNull.Value, detailRow("U_VATCODE")))
            cmdInsertDetail.Parameters.AddWithValue("@U_BUKRS", If(IsDBNull(detailRow("U_BUKRS")), DBNull.Value, detailRow("U_BUKRS")))
            cmdInsertDetail.Parameters.AddWithValue("@U_MWSKZ", If(IsDBNull(detailRow("U_MWSKZ")), DBNull.Value, detailRow("U_MWSKZ")))
            cmdInsertDetail.Parameters.AddWithValue("@U_BELNR", If(IsDBNull(detailRow("U_BELNR")), DBNull.Value, detailRow("U_BELNR")))
            cmdInsertDetail.Parameters.AddWithValue("@U_FA_DESC", If(IsDBNull(detailRow("U_FA_DESC")), DBNull.Value, detailRow("U_FA_DESC")))
            cmdInsertDetail.Parameters.AddWithValue("@U_FA_QTY", detailRow("U_FA_QTY"))
            cmdInsertDetail.Parameters.AddWithValue("@U_FA_USE", If(IsDBNull(detailRow("U_FA_USE")), DBNull.Value, detailRow("U_FA_USE")))
            cmdInsertDetail.Parameters.AddWithValue("@U_GatherMark", If(IsDBNull(detailRow("U_GatherMark")), DBNull.Value, detailRow("U_GatherMark")))
            cmdInsertDetail.Parameters.AddWithValue("@U_ConsolidQty", detailRow("U_ConsolidQty"))
            cmdInsertDetail.Parameters.AddWithValue("@CreateBy", CommUtil.GetCurrentUserId())

            cmdInsertDetail.ExecuteNonQuery()
        Next

        ' ===== 第三步：更新 jtdb.jMGUIAP 的 MDRPostStatus =====
        Dim sqlUpdateStatus As String = "
            UPDATE jMGUIAP
            SET MDRPostStatus = 'P',
                MDRPostDate = GETDATE()
            WHERE jID = @jID
        "

        Dim cmdUpdateStatus As New SqlCommand(sqlUpdateStatus, jtdbConn)
        cmdUpdateStatus.Parameters.AddWithValue("@jID", jID)
        cmdUpdateStatus.ExecuteNonQuery()

        ' 提交交易
        trans.Commit()

        Return mdrHeaderID

    Catch ex As Exception
        ' 回滾交易
        If trans IsNot Nothing Then
            Try
                trans.Rollback()
            Catch
            End Try
        End If

        ' 更新錯誤狀態
        If jtdbConn IsNot Nothing AndAlso jtdbConn.State = ConnectionState.Open Then
            Try
                Dim sqlUpdateError As String = "
                    UPDATE jMGUIAP
                    SET MDRPostStatus = 'E',
                        MDRErrMsg = @ErrorMsg
                    WHERE jID = @jID
                "
                Dim cmdUpdateError As New SqlCommand(sqlUpdateError, jtdbConn)
                cmdUpdateError.Parameters.AddWithValue("@jID", jID)
                cmdUpdateError.Parameters.AddWithValue("@ErrorMsg", ex.Message.Substring(0, Math.Min(500, ex.Message.Length)))
                cmdUpdateError.ExecuteNonQuery()
            Catch
            End Try
        End If

        ' 記錄錯誤日誌
        System.Diagnostics.Debug.WriteLine("WriteMDRData Error: " & ex.Message)

        Throw New Exception("MDR 資料同步失敗：" & ex.Message, ex)

    Finally
        If jtdbConn IsNot Nothing AndAlso jtdbConn.State = ConnectionState.Open Then
            jtdbConn.Close()
        End If
        If mdrConn IsNot Nothing AndAlso mdrConn.State = ConnectionState.Open Then
            mdrConn.Close()
        End If
    End Try
End Function
